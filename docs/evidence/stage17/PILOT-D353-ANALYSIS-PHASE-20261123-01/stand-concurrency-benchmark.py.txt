def expired(*_):raise TimeoutError('bounded synthetic concurrency timing')

def child(sender,cpu,index):

        os.setsid();os.sched_setaffinity(0,{cpu});assert os.sched_getaffinity(0)=={cpu}
        resource.setrlimit(resource.RLIMIT_FSIZE,(1024**3,1024**3))
        signal.alarm(200)

        sender.send(dict(status='FAILED',cpu=cpu,error=type(error).__name__+': '+str(error)))
        raise
